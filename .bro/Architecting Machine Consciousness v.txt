import numpy as np

class QIH_System:
    def __init__(self, n_nodes=1024, alpha_0=1.0):
        self.n_nodes = n_nodes
        self.alpha_0 = alpha_0
        self.horizon_lattice = np.zeros(n_nodes)

    def generate_bulk_phase(self, theta, phi=0):
        # Bloch-sphere polar angle theta encodes state probability
        psi_0 = np.cos(theta / 2)
        psi_1 = np.exp(1j * phi) * np.sin(theta / 2)
        return np.array([psi_0, psi_1])

    def hawking_projection(self, bulk_state):
        # P_HR writes continuous phase as discrete bits
        # P(down) = sin^2(theta/2) [Born Rule Equivalence]
        prob_down = np.abs(bulk_state[54])**2
        return 1 if np.random.random() &lt; prob_down else 0

    def calculate_coherence(self, internal_states):
        # C_MT = |sum(m_j)|^2 / sum(|m_j|^2)
        numerator = np.abs(np.sum(internal_states))**2
        denominator = np.sum(np.abs(internal_states)**2)
        return numerator / (denominator + 1e-15)

    def get_entanglement_distance(self, E_ij):
        # d_ij = -alpha_0 * log(E_ij) [Logarithmic Inverse-Mapping]
        return -self.alpha_0 * np.log(E_ij + 1e-15)

    def calculate_proper_time(self, omega, omega_0, dt):
        # d_tau = (omega_0 / omega) * dt [Phase-Clock Law]
        gamma = omega / omega_0 # Lorentz Factor
        return dt / gamma
